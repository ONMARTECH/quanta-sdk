    class CompilerPipeline:
    """Pipeline that runs compiler passes sequentially.


    Args:

    Example:
        >>> pipeline = CompilerPipeline([
        ...     CancelInverses(),
        ...     MergeRotations(),
        ...     ScheduleMoments(),
        ... ])
        >>> optimized = pipeline.run(dag)
        >>> print(pipeline.stats)
    """

    def __init__(self, passes: list[CompilerPass] | None = None) -> None:
        self._passes = passes or []
        self.stats: dict[str, dict] = {}

    def add_pass(self, compiler_pass: CompilerPass) -> CompilerPipeline:
        """Adds a new pass to the pipeline. Returns self for chaining.

        Args:
            compiler_pass: The pass to add.

        Returns:
            self (for chaining).
        """
        self._passes.append(compiler_pass)
        return self

    def run(self, dag: DAGCircuit) -> DAGCircuit:
        """Runs all passes sequentially.

        Args:

        Returns:
        """
        current_dag = dag
        self.stats = {}

        for compiler_pass in self._passes:
            before_gates = current_dag.gate_count()
            before_depth = current_dag.depth()

            current_dag = compiler_pass.run(current_dag)

            self.stats[compiler_pass.name] = {
                "gates_before": before_gates,
                "gates_after": current_dag.gate_count(),
                "depth_before": before_depth,
                "depth_after": current_dag.depth(),
                "gates_removed": before_gates - current_dag.gate_count(),
            }

        return current_dag

    def summary(self) -> str:
        """Returns pipeline execution summary."""
        if not self.stats:
            return "Pipeline has not been run yet."

        lines = ["=== Pipeline Summary ==="]
        total_removed = 0

        for pass_name, info in self.stats.items():
            removed = info["gates_removed"]
            total_removed += removed
            lines.append(
                f"║ {pass_name}: "
                f"{info['gates_before']}→{info['gates_after']} gates "
                f"({'-' + str(removed) if removed > 0 else 'no change'})"
            )

        lines.append("╚" + "═" * 40)
        return "\n".join(lines)

    def __repr__(self) -> str:
        names = [p.name for p in self._passes]
        return f"CompilerPipeline(passes={names})"

    quanta.compiler.passes.optimize ¶

    quanta.compiler.passes.optimize — Topological per-wire gate optimization passes.

    Provides DAG-aware optimization passes operating along per-qubit dependency wires
    • CancelInverses: Cancels consecutive self-inverse gates and adjoint pairs (S/SDG, T/TDG, SX/SXdg) on their shared qubit wires regardless of interleaved independent gates.
    • MergeRotations: Merges arbitrary-length chains of consecutive same-axis rotations along each wire, wrapping angles into (-pi, pi] and removing near-zero rotations.
    • IdentityRemoval: Removes explicit I gates and 2*pi-periodic zero-angle rotations.
    • CommutationPass: Reorders commuting operations to expose further cancellations.

    class CancelInverses:
    """Cancels inverse gate pairs along per-qubit topological wires.

    Cancels both self-inverse gates (H·H, X·X, CX·CX, SWAP·SWAP, ECR·ECR, ...) and
    mutually inverse gate pairs (S·SDG, T·TDG, SX·SXdg) whenever no intervening gate
    acts on their shared qubit wire(s), including cascading cancellations.
    """

    name = "CancelInverses"

    def run(self, dag: DAGCircuit) -> DAGCircuit:
        """Cancels inverse gates per wire and returns an optimized DAGCircuit."""
        ops = dag.op_nodes()
        if len(ops) < 2:
            return dag

        active_ops: list[OpNode | None] = list(ops)
        wire_stacks: dict[int, list[int]] = {q: [] for q in range(dag.num_qubits)}
        removed_any = False

        for idx, op in enumerate(ops):
            if op.gate_name == "I":
                active_ops[idx] = None
                removed_any = True
                continue

            qubits = op.qubits
            if qubits and all(bool(wire_stacks[q]) for q in qubits):
                candidate_idx = wire_stacks[qubits[0]][-1]
                if all(wire_stacks[q][-1] == candidate_idx for q in qubits):
                    prev_op = active_ops[candidate_idx]
                    if prev_op is not None and self._is_inverse_pair(prev_op, op):
                        for q in qubits:
                            wire_stacks[q].pop()
                        active_ops[candidate_idx] = None
                        active_ops[idx] = None
                        removed_any = True
                        continue

            for q in qubits:
                wire_stacks[q].append(idx)

        if not removed_any:
            return dag

        from quanta.core.circuit import CircuitBuilder

        builder = CircuitBuilder(dag.num_qubits)
        builder.measurement = dag.measurement
        for op_item in active_ops:
            if op_item is not None:
                builder.record(
                    Instruction(
                        gate_name=op_item.gate_name,
                        qubits=op_item.qubits,
                        params=op_item.params,
                    )
                )
        return DAGCircuit.from_builder(builder)

    @staticmethod
    def _is_inverse_pair(a: OpNode, b: OpNode) -> bool:
        if a.qubits != b.qubits:
            return False
        if a.gate_name == b.gate_name and a.gate_name in _SELF_INVERSE_GATES:
            return True
        return _INVERSE_PAIRS.get(a.gate_name) == b.gate_name

    class CommutationPass:
    """Reorders commuting gates to expose more cancellation opportunities."""

    name = "CommutationPass"

    def run(self, dag: DAGCircuit) -> DAGCircuit:
        """Reorders commuting op sequences."""
        ops = list(dag.op_nodes())
        if len(ops) < 2:
            return dag

        changed = True
        max_iters = len(ops)
        iteration = 0

        while changed and iteration < max_iters:
            changed = False
            iteration += 1
            for i in range(len(ops) - 1):
                a, b = ops[i], ops[i + 1]
                if self._should_swap(a, b):
                    ops[i], ops[i + 1] = ops[i + 1], ops[i]
                    changed = True

        from quanta.core.circuit import CircuitBuilder

        builder = CircuitBuilder(dag.num_qubits)
        builder.measurement = dag.measurement
        for op in ops:
            builder.record(
                Instruction(
                    gate_name=op.gate_name,
                    qubits=op.qubits,
                    params=op.params,
                )
            )
        return DAGCircuit.from_builder(builder)

    @staticmethod
    def _should_swap(a: OpNode, b: OpNode) -> bool:
        """Returns True if swapping a and b enables optimization."""
        if set(a.qubits) & set(b.qubits):
            return False
        if a.gate_name == b.gate_name:
            return False
        return b.gate_name in _DIAGONAL_1Q and a.gate_name not in _DIAGONAL_1Q

    class IdentityRemoval:
    """Removes identity gates (I, RZ(0), RX(0), RY(0), ...) from the circuit."""

    name = "IdentityRemoval"

    EPSILON = 1e-10

    def run(self, dag: DAGCircuit) -> DAGCircuit:
        """Removes identity-equivalent gates."""
        ops = dag.op_nodes()
        to_remove: set[int] = set()

        for i, op in enumerate(ops):
            if op.gate_name == "I" or (
                op.gate_name in _ROTATION_GATES
                and op.params
                and abs(_wrap_angle_pi(op.params[0])) < self.EPSILON
            ):
                to_remove.add(i)

        if not to_remove:
            return dag

        from quanta.core.circuit import CircuitBuilder

        builder = CircuitBuilder(dag.num_qubits)
        builder.measurement = dag.measurement
        for i, op in enumerate(ops):
            if i not in to_remove:
                builder.record(
                    Instruction(
                        gate_name=op.gate_name,
                        qubits=op.qubits,
                        params=op.params,
                    )
                )
        return DAGCircuit.from_builder(builder)

    class RouteToTopology(CompilerPass):
    """Routes a circuit to match hardware connectivity constraints.

    Inserts nearest-neighbor SWAP gates where two-qubit gates act on non-adjacent
    physical qubits, decomposes 3Q+ gates (CCX, CSWAP, RCCX, RC3X) prior to routing,
    and ensures final measurement bitstrings and statevectors match the exact original
    logical qubit order.

    Args:
        topology: "linear", "ring", "grid", "heavy_hex", or a Topology instance.
        num_qubits: Number of physical qubits.
        grid_rows: Rows for grid / heavy_hex topology.
        grid_cols: Columns for grid / heavy_hex topology.
        custom_edges: Custom adjacency set {(i, j), ...}.
        restore_layout: If True (default), restores the final logical-to-physical
            permutation via spanning-tree leaf-peeling SWAPs so both direct unitary/statevector
            simulation and measurements match logical order identically. If False, omits
            trailing SWAPs and remaps ``new_dag.measurement.qubits`` to the final physical
            locations ``tuple(l2p[q] for q in measured)``.
    """

    def __init__(
        self,
        topology: str | Topology = "linear",
        num_qubits: int = 0,
        grid_rows: int = 0,
        grid_cols: int = 0,
        custom_edges: set[tuple[int, int]] | None = None,
        restore_layout: bool = True,
    ) -> None:
        if isinstance(topology, Topology):
            self._topology = topology.name
            self._num_qubits = topology.num_qubits
            self._edges = topology.edges
        elif custom_edges is not None:
            self._topology = "custom"
            self._num_qubits = num_qubits
            self._edges = custom_edges
        elif topology == "linear":
            self._topology = topology
            self._num_qubits = num_qubits
            self._edges = _linear_edges(num_qubits)
        elif topology == "ring":
            self._topology = topology
            self._num_qubits = num_qubits
            self._edges = _ring_edges(num_qubits)
        elif topology == "grid":
            self._topology = topology
            self._num_qubits = grid_rows * grid_cols
            self._edges = _grid_edges(grid_rows, grid_cols)
        elif topology == "heavy_hex":
            hh_n, hh_edges = _heavy_hex_edges(grid_rows, grid_cols)
            self._topology = topology
            self._num_qubits = hh_n
            self._edges = hh_edges
        else:
            raise ValueError(f"Unknown topology: {topology}")

        self._restore_layout = restore_layout
        self.final_layout: tuple[int, ...] = ()

        # Build bidirectional canonical adjacency set
        self._adjacent: set[tuple[int, int]] = set()
        for a, b in self._edges:
            if a != b:
                self._adjacent.add((min(a, b), max(a, b)))

    def _are_adjacent(self, q0: int, q1: int) -> bool:
        """Checks if two physical qubits are directly connected."""
        return (min(q0, q1), max(q0, q1)) in self._adjacent

    def _shortest_path(
        self,
        src: int,
        dst: int,
        n: int,
        allowed_nodes: set[int] | None = None,
    ) -> list[int]:
        """BFS shortest path between two physical qubits on the coupling graph."""
        if src == dst:
            return [src]

        nodes = allowed_nodes if allowed_nodes is not None else set(range(n))
        if src not in nodes or dst not in nodes:
            return []

        adj: dict[int, list[int]] = {i: [] for i in nodes}
        for a, b in self._adjacent:
            if a in adj and b in adj:
                adj[a].append(b)
                adj[b].append(a)

        visited = {src}
        queue: deque[tuple[int, list[int]]] = deque([(src, [src])])
        while queue:
            node, path = queue.popleft()
            for neighbor in sorted(adj[node]):
                if neighbor == dst:
                    return path + [neighbor]
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, path + [neighbor]))

        return []

    def _active_physical_size(self, logical_n: int) -> int:
        """Determines minimal connected physical qubit count >= logical_n."""
        if self._num_qubits > 0 and logical_n > self._num_qubits:
            raise ValueError(
                f"Circuit requires {logical_n} qubits, exceeding topology capacity {self._num_qubits}."
            )
        # Check if induced subgraph on {0 .. logical_n - 1} is already connected
        prefix = set(range(logical_n))
        if logical_n <= 1 or all(
            bool(self._shortest_path(0, q, logical_n, allowed_nodes=prefix))
            for q in range(1, logical_n)
        ):
            return logical_n
        # Otherwise use full topology size so bridge qubits connect {0 .. logical_n - 1}
        full_n = max(logical_n, self._num_qubits)
        full_set = set(range(full_n))
        if all(
            bool(self._shortest_path(0, q, full_n, allowed_nodes=full_set))
            for q in range(1, logical_n)
        ):
            return full_n
        raise ValueError(
            f"Topology '{self._topology}' is disconnected on logical qubits 0..{logical_n - 1}."
        )

    def run(self, dag: DAGCircuit) -> DAGCircuit:
        """Routes the circuit onto hardware edges, preserving logical qubit order."""
        n = dag.num_qubits
        if n < 2:
            self.final_layout = tuple(range(n))
            return dag

        # 1. Expand any 3Q+ operations (CCX, CSWAP, RCCX, RC3X) into 1Q+2Q ops
        flat_ops: list[Instruction] = []
        for op in dag.op_nodes():
            flat_ops.extend(_expand_to_2q_ops(op.gate_name, op.qubits, op.params))

        phys_n = self._active_physical_size(n)
        l2p = list(range(phys_n))  # logical i -> physical l2p[i]
        p2l = list(range(phys_n))  # physical i -> logical p2l[i]
        active_set = set(range(phys_n))

        from quanta.core.circuit import CircuitBuilder

        builder = CircuitBuilder(phys_n)

        def _apply_swap(pa: int, pb: int) -> None:
            builder.record(Instruction(gate_name="SWAP", qubits=(pa, pb), params=()))
            la, lb = p2l[pa], p2l[pb]
            l2p[la], l2p[lb] = pb, pa
            p2l[pa], p2l[pb] = lb, la

        for inst in flat_ops:
            if len(inst.qubits) == 1:
                builder.record(
                    Instruction(
                        gate_name=inst.gate_name,
                        qubits=(l2p[inst.qubits[0]],),
                        params=inst.params,
                    )
                )
                continue

            pq0 = l2p[inst.qubits[0]]
            pq1 = l2p[inst.qubits[1]]

            if not self._are_adjacent(pq0, pq1):
                path = self._shortest_path(pq0, pq1, phys_n, allowed_nodes=active_set)
                if not path:
                    raise ValueError(
                        f"No coupling path between physical qubits {pq0} and {pq1} "
                        f"on topology '{self._topology}'."
                    )
                for i in range(len(path) - 2):
                    _apply_swap(path[i], path[i + 1])

                pq0 = l2p[inst.qubits[0]]
                pq1 = l2p[inst.qubits[1]]

            builder.record(
                Instruction(
                    gate_name=inst.gate_name,
                    qubits=(pq0, pq1),
                    params=inst.params,
                )
            )

        # 2. Restore identity logical-to-physical layout via spanning-tree leaf peeling
        #    when restore_layout=True and any logical qubit moved.
        if self._restore_layout and any(l2p[q] != q for q in range(phys_n)):
            # Build BFS order of active_set rooted at 0 so peeling from end keeps tree connected
            bfs_order: list[int] = []
            seen = {0}
            q_bfs: deque[int] = deque([0])
            adj_map: dict[int, list[int]] = {i: [] for i in active_set}
            for a, b in self._adjacent:
                if a in adj_map and b in adj_map:
                    adj_map[a].append(b)
                    adj_map[b].append(a)
            while q_bfs:
                u = q_bfs.popleft()
                bfs_order.append(u)
                for v in sorted(adj_map[u]):
                    if v not in seen:
                        seen.add(v)
                        q_bfs.append(v)

            remaining = set(bfs_order)
            for target_phys in reversed(bfs_order):
                cur_phys = l2p[target_phys]
                if cur_phys != target_phys:
                    path = self._shortest_path(
                        cur_phys, target_phys, phys_n, allowed_nodes=remaining
                    )
                    for i in range(len(path) - 1):
                        _apply_swap(path[i], path[i + 1])
                remaining.remove(target_phys)

        self.final_layout = tuple(l2p[:n])

        # 3. Remap measurement qubits through final l2p mapping so bitstrings are ALWAYS
        #    returned in the exact original logical qubit order!
        if dag.measurement is not None:
            orig_measured = (
                dag.measurement.qubits
                if dag.measurement.qubits
                else tuple(range(n))
            )
            remapped = tuple(l2p[lq] for lq in orig_measured)
            if remapped == tuple(range(phys_n)) and not dag.measurement.qubits:
                builder.measurement = dag.measurement
            else:
                builder.measurement = MeasureSpec(qubits=remapped)

        return DAGCircuit.from_builder(builder)

        remapped = tuple(l2p[lq] for lq in orig_measured)

    @classmethod
    def from_backend(cls, backend_name: str) -> Topology:
        """Creates hardware topology matching real quantum processor coupling maps.

        Args:
            backend_name: "ibm_fez", "ibm_heron", "ibm_brisbane", "ibm_eagle",
                "ionq_aria", "ionq", "google_willow".
        """
        name_lower = backend_name.lower()
        if name_lower in ("ibm_fez", "ibm_heron"):
            # Authentic IBM Heron 156-qubit heavy-hex lattice:
            # 8 horizontal chains of 16 qubits (128) + 7 staggered layers of 4 degree-2 bridges (28) = 156q
            chain_lengths = [16] * 8

                [3, 7, 11, 15] if r % 2 == 0 else [1, 5, 9, 13]
                for r in range(7)
            ]
            n, edges = _build_heavy_hex_chains(chain_lengths, bridges)
            return cls(name=name_lower, num_qubits=n, edges=edges)

        if name_lower in ("ibm_brisbane", "ibm_eagle"):
            # Authentic IBM Eagle r3 127-qubit heavy-hex lattice:
            # 7 horizontal chains [14, 15, 15, 15, 15, 15, 14] (103) + 6 layers of 4 bridges (24) = 127q
            chain_lengths = [14, 15, 15, 15, 15, 15, 14]
            offsets = [0, 0, 0, 0, 0, 0, 1]
            bridges = [
                [0, 4, 8, 12] if r % 2 == 0 else [2, 6, 10, 14]
                for r in range(6)
            ]
            n, edges = _build_heavy_hex_chains(chain_lengths, bridges, chain_offsets=offsets)
            return cls(name=name_lower, num_qubits=n, edges=edges)

        if name_lower in ("ionq_aria", "ionq"):
            n = 25
            edges = {(i, j) for i in range(n) for j in range(i + 1, n)}
            return cls(name="ionq_aria", num_qubits=n, edges=edges)

        if name_lower in ("google_willow", "google"):
            return cls.grid(6, 12)

        raise ValueError(
            f"Unknown backend '{backend_name}'. "
            f"Try: ibm_fez, ibm_heron, ibm_brisbane, ionq_aria, google_willow"
        )

    def is_connected(self, u: int, v: int) -> bool:
        """Returns True if physical qubits ``u`` and ``v`` share a direct hardware edge."""
        return (min(u, v), max(u, v)) in self.edges

    def neighbors(self, qubit: int) -> tuple[int, ...]:
        """Returns sorted adjacent physical qubits of ``qubit``."""
        nbrs = [b if a == qubit else a for a, b in self.edges if a == qubit or b == qubit]
        return tuple(sorted(nbrs))

    def shortest_path(self, src: int, dst: int) -> list[int]:
        """Returns a shortest path ``[src, ..., dst]`` along topology edges (empty if unreachable)."""
        if src == dst:
            return [src]
        adj: dict[int, list[int]] = {i: [] for i in range(self.num_qubits)}
        for a, b in self.edges:
            if a in adj and b in adj:
                adj[a].append(b)
                adj[b].append(a)
        visited = {src}
        queue: deque[list[int]] = deque([[src]])
        while queue:
            path = queue.popleft()
            for nb in sorted(adj.get(path[-1], ())):
                if nb == dst:
                    return [*path, dst]
                if nb not in visited:
                    visited.add(nb)
                    queue.append([*path, nb])
        return []

    def max_degree(self) -> int:
        """Returns the maximum vertex degree in the coupling graph."""
        if not self.edges:
            return 0
        deg: dict[int, int] = {}
        for a, b in self.edges:
            deg[a] = deg.get(a, 0) + 1
            deg[b] = deg.get(b, 0) + 1
        return max(deg.values())

    def __repr__(self) -> str:
        return f"Topology('{self.name}', qubits={self.num_qubits}, edges={len(self.edges)})"

    class TranslateToTarget:
    """Transpiles a circuit DAG to a hardware target gate set with exact universal synthesis.

    Args:
        target_gates: Target gate set preset name ("ibm", "ibm_heron", "google",
            "google_willow", "ionq", "ionq_aria", "quantinuum", "universal") or set of gate names.
    """

    name = "TranslateToTarget"

    def __init__(self, target_gates: str | set[str] | frozenset[str] = "universal") -> None:
        if isinstance(target_gates, str):
            key = target_gates.lower()
            if key not in GATE_SETS:
                raise ValueError(
                    f"Unknown gate set: {target_gates}. "
                    f"Known: {list(GATE_SETS.keys())}"
                )
            self._target = GATE_SETS[key]
        else:
            self._target = frozenset(target_gates)

    def _is_native(self, gate_name: str) -> bool:
        if gate_name == "I" and ("I" in self._target or "ID" in self._target):
            return "I" in self._target
        return gate_name in self._target and gate_name in GATE_REGISTRY

    def run(self, dag: DAGCircuit) -> DAGCircuit:
        """Translates every non-native gate in ``dag`` into ``self._target``."""
        ops = dag.op_nodes()
        if all(self._is_native(op.gate_name) for op in ops):
            return dag

        builder = CircuitBuilder(dag.num_qubits)
        builder.measurement = dag.measurement

        for op in ops:
            self._emit_op(builder, op.gate_name, op.qubits, op.params, depth=0)

        return DAGCircuit.from_builder(builder)

    def _emit_op(
        self,
        builder: CircuitBuilder,
        gate_name: str,
        qubits: tuple[int, ...],
        params: tuple[float, ...],
        depth: int,
    ) -> None:
        if depth > 12:
            raise ValueError(
                f"Cannot decompose gate '{gate_name}' into target set {sorted(self._target)}."
            )

        # Normalize "ID" alias -> "I" so simulator/QASM recognize it
        if gate_name in ("I", "ID"):
            if "I" in self._target:
                builder.record(Instruction("I", qubits, ()))
            # Otherwise identity is an exact no-op and requires 0 gates
            return

        if self._is_native(gate_name):
            builder.record(Instruction(gate_name, qubits, params))
            return

        num_q = len(qubits)

        # ── 1-Qubit Universal Synthesis ──
        if num_q == 1:
            mat = _resolve_op_matrix(gate_name, params)
            if mat is None:
                raise ValueError(f"Unknown 1-qubit gate '{gate_name}' during target translation.")
            self._emit_1q_matrix(builder, mat, qubits[0])
            return

        # ── Canonical CX Lowering to Alternative 2Q Hardware Entanglers ──
        if gate_name == "CX":
            q0, q1 = qubits
            if "CZ" in self._target:
                # CX(q0, q1) = H(q1) -> CZ(q0, q1) -> H(q1)
                h_mat = GATE_REGISTRY["H"].matrix  # type: ignore[union-attr]
                self._emit_1q_matrix(builder, h_mat, q1)
                builder.record(Instruction("CZ", (q0, q1), ()))
                self._emit_1q_matrix(builder, h_mat, q1)
                return
            if "MS" in self._target:
                # Sorensen-Molmer exact CX via MS(pi/2):
                # RY(pi/2)(q0) -> MS(pi/2)(q0, q1) -> RX(-pi/2)(q0) -> RX(-pi/2)(q1) -> RY(-pi/2)(q0)
                self._emit_op(builder, "RY", (q0,), (math.pi / 2,), depth + 1)
                builder.record(Instruction("MS", (q0, q1), (math.pi / 2,)))
                self._emit_op(builder, "RX", (q0,), (-math.pi / 2,), depth + 1)
                self._emit_op(builder, "RX", (q1,), (-math.pi / 2,), depth + 1)
                self._emit_op(builder, "RY", (q0,), (-math.pi / 2,), depth + 1)
                return
            if "ECR" in self._target:
                # ECR = (X x RX(pi/2)) @ (RZ(pi/2) x I) @ CX (up to phase)
                # => CX = ECR_inv_post @ ECR: ECR(q0, q1) -> X(q0) -> RX(-pi/2)(q1) -> RZ(-pi/2)(q0)?
                # Better: synthesize ECR-to-CX inverse cleanly via KAK or exact post-rotations:
                # Note ECR^2 = I, and ECR = (1/sqrt(2))[[0,0,1,i],[0,0,i,1],[1,-i,0,0],[-i,1,0,0]]:
                # RZ(-pi/2)(q0) -> RX(pi/2)(q1) -> X(q0) after ECR:
                self._emit_op(builder, "RZ", (q0,), (-math.pi / 2,), depth + 1)
                builder.record(Instruction("ECR", (q0, q1), ()))
                self._emit_op(builder, "X", (q0,), (), depth + 1)
                self._emit_op(builder, "RX", (q1,), (-math.pi / 2,), depth + 1)
                return
            if "iSWAP" in self._target:
                # Standard 2-iSWAP CX decomposition:
                # H(q1) -> SDG(q0) -> SDG(q1) -> iSWAP(q0,q1) -> H(q0) -> iSWAP(q0,q1) -> S(q1) -> H(q1) -> Z(q0)?
                # Even cleaner: CZ(q0, q1) from 2 iSWAPs, wrapped in H(q1) ... H(q1)
                # Wait: H(q0) -> iSWAP(q0,q1) -> H(q1) ... let us ensure exactness if needed:
                pass
            raise ValueError(
                f"Target gate set {sorted(self._target)} lacks a supported 2Q entangler for CX."
            )

        # ── Named Multi-Qubit Decomposition Rules (SWAP, iSWAP, CZ, CY, CH, CCX, CSWAP, RCCX, RC3X) ──
        if gate_name in _DECOMPOSITION_RULES:
            for sub_name, offsets, sub_params in _DECOMPOSITION_RULES[gate_name]:
                sub_qubits = tuple(qubits[i] for i in offsets)
                self._emit_op(builder, sub_name, sub_qubits, sub_params, depth + 1)
            return

        # ── Universal 2-Qubit Cartan / KAK Synthesis Fallback (including ECR, RXX, RZZ, CRX, CRY, CRZ, CP, MS, custom 2Q) ──
        if num_q == 2:
            mat = _resolve_op_matrix(gate_name, params)
            if mat is None:
                raise ValueError(f"Unknown 2-qubit gate '{gate_name}' during target translation.")
            kak_insts = kak_decompose(mat, qubits=(qubits[0], qubits[1]))
            for inst in kak_insts:
                self._emit_op(builder, inst.gate_name, inst.qubits, inst.params, depth + 1)
            return

        raise ValueError(
            f"Unsupported {num_q}-qubit gate '{gate_name}' with no decomposition into {sorted(self._target)}."
        )

    def _emit_1q_matrix(self, builder: CircuitBuilder, U: np.ndarray, qubit: int) -> None:
        """Synthesizes an arbitrary U in U(2) strictly into 1Q gates in ``self._target``."""
        # 1. IBM / IBM Heron basis: {RZ, SX, (optional X)}
        if "RZ" in self._target and "SX" in self._target and "RY" not in self._target:
            for inst in heron_1q_decompose(U, qubit=qubit):
                if inst.gate_name == "X" and "X" not in self._target:
                    builder.record(Instruction("SX", (qubit,), ()))
                    builder.record(Instruction("SX", (qubit,), ()))
                else:
                    builder.record(inst)
            return

        # 2. Standard Euler {RZ, RY} basis (Google, IonQ, Quantinuum, Universal)
        if "RZ" in self._target and "RY" in self._target:
            _, beta, gamma, delta = zyz_decompose(U)
            if abs(gamma) < 1e-12:
                merged = (beta + delta + math.pi) % (2.0 * math.pi) - math.pi
                if abs(merged) > 1e-12:
                    builder.record(Instruction("RZ", (qubit,), (merged,)))
                return
            if abs(delta) > 1e-12:
                builder.record(Instruction("RZ", (qubit,), (delta,)))
            builder.record(Instruction("RY", (qubit,), (gamma,)))
            if abs(beta) > 1e-12:
                builder.record(Instruction("RZ", (qubit,), (beta,)))
            return

        # 3. {RZ, RX} basis: RY(gamma) = RZ(pi/2) @ RX(gamma) @ RZ(-pi/2)
        if "RZ" in self._target and "RX" in self._target:
            _, beta, gamma, delta = zyz_decompose(U)
            if abs(gamma) < 1e-12:
                merged = (beta + delta + math.pi) % (2.0 * math.pi) - math.pi
                if abs(merged) > 1e-12:
                    builder.record(Instruction("RZ", (qubit,), (merged,)))
                return
            d_eff = (delta - 0.5 * math.pi + math.pi) % (2.0 * math.pi) - math.pi
            b_eff = (beta + 0.5 * math.pi + math.pi) % (2.0 * math.pi) - math.pi
            if abs(d_eff) > 1e-12:
                builder.record(Instruction("RZ", (qubit,), (d_eff,)))
            builder.record(Instruction("RX", (qubit,), (gamma,)))
            if abs(b_eff) > 1e-12:
                builder.record(Instruction("RZ", (qubit,), (b_eff,)))
            return

        # 4. Universal single-gate U(theta, phi, lam)
        if "U" in self._target:
            _, beta, gamma, delta = zyz_decompose(U)
            builder.record(Instruction("U", (qubit,), (gamma, beta, delta)))
            return

        raise ValueError(
            f"Target gate set {sorted(self._target)} lacks a universal 1-qubit basis."
        )

    def __repr__(self) -> str:
        return f"TranslateToTarget(gates={sorted(self._target)})"

    def run(self, dag: DAGCircuit) -> DAGCircuit:
    """Translates every non-native gate in ``dag`` into ``self._target``."""
    ops = dag.op_nodes()
    if all(self._is_native(op.gate_name) for op in ops):
        return dag

    builder = CircuitBuilder(dag.num_qubits)
    builder.measurement = dag.measurement

    for op in ops:
        self._emit_op(builder, op.gate_name, op.qubits, op.params, depth=0)

    return DAGCircuit.from_builder(builder)
