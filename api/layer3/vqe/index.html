    class VQEResult:
    """Result of a VQE computation."""

    def __init__(
        self,
        energy: float,
        optimal_params: np.ndarray,
        history: list[float],
    ) -> None:
        self.energy = energy
        self.optimal_params = optimal_params
        self.history = history
        self.num_iterations = len(history)

    def __repr__(self) -> str:
        return (
            f"VQEResult(energy={self.energy:.6f}, "
            f"iterations={self.num_iterations})"
        )

    def summary(self) -> str:
        lines = [
            "=== VQE Result ===",
            f"  Ground state energy: {self.energy:.6f}",
            f"  Iterations: {self.num_iterations}",
            f"  Convergence: {abs(self.history[-1] - self.history[-2]):.2e}"
            if len(self.history) > 1
            else "",
        ]
        return "\n".join(lines)

    def build_hamiltonian_matrix(
    terms: list[tuple[str, float]],
    num_qubits: int,
) -> np.ndarray:
    """Builds the explicit 2^n x 2^n Hermitian matrix from Pauli terms (for exact diag/testing)."""
    dim = 1 << num_qubits
    H_mat = np.zeros((dim, dim), dtype=complex)

    for pauli_str, coeff in terms:
        padded = pauli_str.ljust(num_qubits, "I")
        term = np.array([[1.0]], dtype=complex)
        for ch in padded:
            term = np.kron(term, _PAULI[ch])
        H_mat += coeff * term

    return H_mat

    def vqe(
    num_qubits: int,
    hamiltonian: list[tuple[str, float]],
    layers: int = 2,
    max_iter: int = 200,
    learning_rate: float = 0.1,
    seed: int | None = None,
) -> VQEResult:
    """Variational Quantum Eigensolver using O(m * 2^n) bitwise Pauli expectations."""
    rng = np.random.default_rng(seed)
    compiled_terms = _compile_pauli_masks(hamiltonian, num_qubits)
    basis_indices = np.arange(1 << num_qubits, dtype=np.uint64)

    num_params = 2 * num_qubits * layers
    params = rng.uniform(-np.pi, np.pi, size=num_params)

    history: list[float] = []
    best_energy = float("inf")
    best_params = params.copy()
    shift = np.pi / 2.0

    # Adam moments for reliable convergence within max_iter
    m_state = np.zeros(num_params, dtype=float)
    v_state = np.zeros(num_params, dtype=float)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for iteration in range(max_iter):
        state = _build_ansatz_state(num_qubits, layers, params)
        energy = _eval_compiled_pauli_expectation(
            compiled_terms, state, basis_indices
        )
        history.append(energy)

        if energy < best_energy:
            best_energy = energy
            best_params = params.copy()

        gradients = np.zeros(num_params, dtype=float)
        for i in range(num_params):
            orig = params[i]

            params[i] = orig + shift
            state_plus = _build_ansatz_state(num_qubits, layers, params)
            e_plus = _eval_compiled_pauli_expectation(
                compiled_terms, state_plus, basis_indices
            )

            params[i] = orig - shift
            state_minus = _build_ansatz_state(num_qubits, layers, params)
            e_minus = _eval_compiled_pauli_expectation(
                compiled_terms, state_minus, basis_indices
            )

            params[i] = orig
            gradients[i] = 0.5 * (e_plus - e_minus)

        t = iteration + 1
        m_state = beta1 * m_state + (1.0 - beta1) * gradients
        v_state = beta2 * v_state + (1.0 - beta2) * (gradients**2)
        m_hat = m_state / (1.0 - beta1**t)
        v_hat = v_state / (1.0 - beta2**t)
        params -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        if iteration > 10 and abs(history[-1] - history[-2]) < 1e-9:
            break

    # Final check after last update
    final_state = _build_ansatz_state(num_qubits, layers, params)
    final_energy = _eval_compiled_pauli_expectation(
        compiled_terms, final_state, basis_indices
    )
    if final_energy < best_energy:
        best_energy = final_energy
        best_params = params.copy()

    return VQEResult(
        energy=best_energy,
        optimal_params=best_params,
        history=history,
    )

    def evolve(
    hamiltonian: HamiltonianSpec | list[tuple[str, float]],
    num_qubits: int | None = None,
    time: float = 1.0,
    steps: int = 20,
    initial_state: np.ndarray | None = None,
    order: int | None = None,
    method: str = "exact",
) -> EvolutionResult:
    """Simulates time evolution under a Pauli Hamiltonian."""
    if method in ("trotter", "suzuki") or (order is not None and order > 1):
        return trotter_evolve(
            hamiltonian=hamiltonian,
            num_qubits=num_qubits,
            time=time,
            steps=steps,
            order=order or 2,
            initial_state=initial_state,
        )

    if isinstance(hamiltonian, HamiltonianSpec):
        terms = hamiltonian.terms
        n = hamiltonian.num_qubits
    else:
        terms = hamiltonian
        n = num_qubits or max(len(t[0]) for t in terms)

    compiled = _compile_pauli_masks(terms, n)
    dim = 1 << n
    basis_indices = np.arange(dim, dtype=np.uint64)
    H_mat = _build_matrix(terms, n)
    dt = time / steps

    if initial_state is not None:
        state = initial_state.copy().astype(complex)
    else:
        state = np.zeros(dim, dtype=complex)
        state[0] = 1.0

    U_step = spectral_unitary_evolution(H_mat, dt)
    energy_history: list[float] = []

    for _step in range(steps):
        energy = _eval_compiled_pauli_expectation(compiled, state, basis_indices)
        energy_history.append(energy)
        state = U_step @ state

    final_energy = _eval_compiled_pauli_expectation(compiled, state, basis_indices)
    energy_history.append(final_energy)

    return EvolutionResult(
        final_state=state,
        energy=final_energy,
        energy_history=energy_history,
        time=time,
    )

    def evolve_time_dependent(
    H_func: Callable[[float], np.ndarray],
    num_qubits: int,
    t_span: tuple[float, float],
    steps: int = 100,
    initial_state: np.ndarray | None = None,
    order: int = 4,
) -> EvolutionResult:
    """Simulates time evolution under a time-dependent Hamiltonian H(t)."""
    t0, t1 = t_span
    total_time = t1 - t0
    dt = total_time / steps
    dim = 1 << num_qubits

    if initial_state is not None:
        state = initial_state.copy().astype(complex)
    else:
        state = np.zeros(dim, dtype=complex)
        state[0] = 1.0

    energy_history: list[float] = []
    current_time = t0

    for _step in range(steps):
        H_now = H_func(current_time)
        energy_history.append(float(np.real(state.conj() @ H_now @ state)))
        U = magnus_step(H_func, current_time, dt, order=order)
        state = U @ state
        current_time += dt

    H_final = H_func(t1)
    final_energy = float(np.real(state.conj() @ H_final @ state))
    energy_history.append(final_energy)

    return EvolutionResult(
        final_state=state,
        energy=final_energy,
        energy_history=energy_history,
        time=total_time,
    )

    def magnus_step(
    H_func: Callable[[float], np.ndarray],
    t: float,
    dt: float,
    order: int = 4,
) -> np.ndarray:
    """Computes unitary time evolution step U(t + dt, t) via Magnus expansion."""
    if order == 2:
        H_mid = H_func(t + dt / 2.0)
        return spectral_unitary_evolution(H_mid, dt)
    if order == 4:
        c1 = 0.5 - np.sqrt(3.0) / 6.0
        c2 = 0.5 + np.sqrt(3.0) / 6.0
        H1 = H_func(t + c1 * dt)
        H2 = H_func(t + c2 * dt)
        comm = H2 @ H1 - H1 @ H2
        omega = -1j * (dt / 2.0) * (H1 + H2) - (dt**2 * np.sqrt(3.0) / 12.0) * comm
        return _matrix_exp(omega)
    raise ValueError(f"Magnus order {order} not supported. Use order 2 or 4.")

    def molecular_hamiltonian(molecule: str) -> HamiltonianSpec:
    """Returns a pre-defined molecular Hamiltonian (H2, LiH, HeH+)."""
    key = molecule.upper().replace(" ", "")
    if key not in _MOLECULES:
        aliases = {"HEH+": "HeH+", "HEH": "HeH+", "H2": "H2", "LIH": "LiH"}

        raise ValueError(
            f"Unknown molecule: {molecule}. Available: {available}"
        )
    return _MOLECULES[key]

    def spectral_unitary_evolution(H: np.ndarray, dt: float) -> np.ndarray:
    """Computes U(dt) = exp(-i * H * dt) via exact Hermitian spectral decomposition."""
    H_herm = (H + H.conj().T) / 2.0
    eigenvalues, eigenvectors = np.linalg.eigh(H_herm)
    phases = np.exp(-1j * eigenvalues * dt)
    return (eigenvectors * phases) @ eigenvectors.conj().T

    def suzuki_trotter_step(
    term_matrices: list[np.ndarray],
    dt: float,
    order: int = 2,
) -> np.ndarray:
    """Computes a single Suzuki-Trotter step unitary matrix (order 1, 2, or 4)."""
    if not term_matrices:
        raise ValueError("term_matrices list cannot be empty.")

    m = len(term_matrices)
    dim = term_matrices[0].shape[0]

    if m == 1:
        return spectral_unitary_evolution(term_matrices[0], dt)

    if order == 1:
        U = np.eye(dim, dtype=complex)
        for H_k in term_matrices:
            U = spectral_unitary_evolution(H_k, dt) @ U
        return U

    if order == 2:
        U = np.eye(dim, dtype=complex)
        for k in range(m - 1):
            U = spectral_unitary_evolution(term_matrices[k], dt / 2.0) @ U
        U = spectral_unitary_evolution(term_matrices[m - 1], dt) @ U
        for k in range(m - 2, -1, -1):
            U = spectral_unitary_evolution(term_matrices[k], dt / 2.0) @ U
        return U

    if order == 4:
        p = 1.0 / (4.0 - 4.0 ** (1.0 / 3.0))
        S2_p = suzuki_trotter_step(term_matrices, p * dt, order=2)
        S2_mid = suzuki_trotter_step(term_matrices, (1.0 - 4.0 * p) * dt, order=2)
        return S2_p @ S2_p @ S2_mid @ S2_p @ S2_p

    raise ValueError(f"Unsupported Trotter order {order}. Must be 1, 2, or 4.")

    """Simulates Suzuki-Trotter evolution in O(steps * m * 2^n) without any O(4^n) matrices."""
    if isinstance(hamiltonian, HamiltonianSpec):
        terms = hamiltonian.terms
        n = hamiltonian.num_qubits
    else:
        terms = hamiltonian
        n = num_qubits or max(len(t[0]) for t in terms)

    compiled = _compile_pauli_masks(terms, n)
    unit_compiled: list[tuple[int, int, complex, float]] = []
    for (pauli_str, coeff), (f_m, z_m, _) in zip(terms, compiled, strict=True):
        y_cnt = pauli_str.ljust(n, "I")[:n].count("Y")
        unit_compiled.append((f_m, z_m, 1j**y_cnt, float(coeff)))

    dim = 1 << n
    basis_indices = np.arange(dim, dtype=np.uint64)
    dt = time / steps

    if initial_state is not None:
        state = initial_state.copy().astype(complex)
    else:
        state = np.zeros(dim, dtype=complex)
        state[0] = 1.0

    energy_history: list[float] = []
    for _step in range(steps):
        energy = _eval_compiled_pauli_expectation(compiled, state, basis_indices)
        energy_history.append(energy)
        state = _trotter_state_step(unit_compiled, dt, order, state, basis_indices)

    final_energy = _eval_compiled_pauli_expectation(compiled, state, basis_indices)
    energy_history.append(final_energy)

    return EvolutionResult(
        final_state=state,
        energy=final_energy,
        energy_history=energy_history,
        time=time,
    )
